runtime_dir = 'workspace/runtime'
all_py = set()
for root, dirs, files in os.walk(runtime_dir):
    for f in files:
        if f.endswith('.py'):
            rel = os.path.relpath(os.path.join(root, f), runtime_dir).replace(os.sep, '/')
            all_py.add(rel)
# Check each has either imports TO it or __main__ block
has_main = set()
for py in all_py:
    try:
        with open(os.path.join(runtime_dir, py)) as fh:
            tree = ast.parse(fh.read())
        for node in ast.walk(tree):
            if isinstance(node, ast.If):

                if isinstance(test, ast.Compare) and any(
