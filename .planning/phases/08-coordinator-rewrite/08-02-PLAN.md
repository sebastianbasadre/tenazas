```python
# ── Core Routing ─────────────────────────────────────────────────


def _get_claimants(query: str) -> list:
    """Run all owns() calls in parallel. Returns list of agent names that claimed.

    Uses ThreadPoolExecutor with 2s timeout — owns() is pure keyword matching (<10ms).
    If an agent's owns() raises or times out, it is silently excluded (no claim).
    """
    claimants = []
    agents = {name: _load_agent(name) for name in AGENT_MODULES}
    loaded = {name: mod for name, mod in agents.items() if mod is not None}
    if not loaded:
        return claimants
    with ThreadPoolExecutor(max_workers=5) as pool:
        futures = {
            pool.submit(mod.owns, query): name
            for name, mod in loaded.items()
        }
        try:
            for future in as_completed(futures, timeout=2):
                name = futures[future]
                try:
                    if future.result():
                        claimants.append(name)
                except Exception:
                    pass  # owns() failure = no claim
        except FutureTimeout:
            pass  # partial results OK
    return claimants


def _run_single(name: str, query: str, correlation_id: str) -> dict:
    """Direct call to one agent. No threading overhead (first-wins strategy).

    Returns {agent_name: (Observation, recs)} or {} on error.
    """
    try:
        agent = _load_agent(name)
        if agent is None:
            return {}
        obs, recs = agent.handle(query, correlation_id=correlation_id)
        return {name: (obs, recs)}
    except Exception as e:
        _log({"action": "agent_error", "agent": name,
              "error": str(e)[:100], "correlation_id": correlation_id})
        return {}


def _run_parallel(claimants: list, query: str, correlation_id: str) -> dict:
    """Call handle() on all claimants in parallel. Timeout = 30s per agent.

    Returns {agent_name: (Observation, recs)} for each agent that responded.
    Timed-out or erroring agents are omitted from results (not fatal).
    """
    results = {}
    with ThreadPoolExecutor(max_workers=len(claimants)) as pool:
        futures = {
            pool.submit(_load_agent(name).handle, query,
                        correlation_id=correlation_id): name
            for name in claimants
            if _load_agent(name) is not None
        }
        try:
            for future in as_completed(futures, timeout=31):
                name = futures[future]
                try:
                    obs, recs = future.result(timeout=30)
                    results[name] = (obs, recs)
                except FutureTimeout:
                    _log({"action": "agent_timeout", "agent": name,
                          "correlation_id": correlation_id})
                except Exception as e:
                    _log({"action": "agent_error", "agent": name,
                          "error": str(e)[:100], "correlation_id": correlation_id})
        except FutureTimeout:
            _log({"action": "parallel_timeout", "correlation_id": correlation_id,
                  "partial_results": list(results.keys())})
    return results


def _aggregate(results: dict) -> str:
    """Aggregate multiple agent results using emoji-labeled sections.

    results: {agent_name: (Observation, list[Recommendation])}
             OR {agent_name: str}   (backward compat with old TestCoordinator tests)

    Returns newline-joined sections. Skips agents with empty summaries.
    Returns None if all summaries are empty.
    """
    parts = []
    for name, value in results.items():
        # Support both new (Observation, recs) tuples and legacy str values
        if isinstance(value, tuple):
            obs, _recs = value
            summary = obs.summary if obs and obs.summary else None
        else:
            summary = value if value else None
        if not summary:
            continue
        emoji = DOMAIN_EMOJI.get(name, "•")
        label = name.replace("_", " ").title()
        parts.append(f"{emoji} {label}: {summary}")
    return "\n".join(parts) if parts else None


def _do_skill_fallback(query: str, correlation_id: str) -> dict:
    """Zero-claim fallback: try skill_executor for skill: categories, signal LLM for chat.

    Does NOT call orchestrator.route_query() — per D-01 constraint.
    Returns {handled, output, source}.
    """
    from classifier import classify_query
    from skill_executor import execute_skill

    category = classify_query(query)
    if category.startswith("skill:"):
        result = execute_skill(category, query)
        _log({"action": "skill_fallback", "category": category,
              "success": result.get("success"), "correlation_id": correlation_id})
        return {
            "handled": result.get("success", False),
            "output": result.get("output"),
            "source": "skill_executor",
        }
    _log({"action": "llm_fallback", "category": category,
          "correlation_id": correlation_id})
    return {"handled": False, "output": None, "source": "llm_needed"}


def route(query: str) -> dict:
    """Route query to appropriate agent(s) and aggregate responses.

    Execution strategies (per D-03):
    - 0 claimants  → skill_fallback (skill_executor or LLM signal)
    - 1 claimant   → first_wins (direct call, no threading)
    - 2+ claimants → parallel (ThreadPoolExecutor, 30s timeout per agent)

    Returns: {query, correlation_id, strategy, agents, responded, aggregated, handled}
    """
    correlation_id = str(uuid.uuid4())[:8]
    _log({"action": "route_start", "correlation_id": correlation_id, "query": query[:80]})
    t0 = now()

    claimants = _get_claimants(query)

    if not claimants:
        strategy = "skill_fallback"
        result = _do_skill_fallback(query, correlation_id)
        elapsed = (now() - t0).total_seconds()
        _log({"action": "route_done", "correlation_id": correlation_id,
              "strategy": strategy, "agents": [], "elapsed": elapsed})
        _save_coord_log({**result, "correlation_id": correlation_id,
                         "strategy": strategy, "query": query,
                         "agents": [], "ts": now().isoformat()})
        return {**result, "correlation_id": correlation_id, "strategy": strategy,
                "query": query, "agents": [], "responded": []}

    elif len(claimants) == 1:
        strategy = "first_wins"
        results = _run_single(claimants[0], query, correlation_id)
    else:
        strategy = "parallel"
        results = _run_parallel(claimants, query, correlation_id)

    if len(results) > 1:
        aggregated = _aggregate(results)
    elif results:
        name = list(results.keys())[0]
        obs, _ = results[name]
        aggregated = obs.summary if obs and obs.summary else None
    else:
        aggregated = None

    elapsed = (now() - t0).total_seconds()
    responded = list(results.keys())
    _log({"action": "route_done", "correlation_id": correlation_id,
          "strategy": strategy, "agents": claimants,
          "responded": responded, "elapsed": elapsed})
    _save_coord_log({"query": query, "correlation_id": correlation_id,
                     "strategy": strategy, "agents": claimants,
                     "responded": responded, "handled": bool(aggregated),
                     "ts": now().isoformat()})
    return {
        "query": query,
        "correlation_id": correlation_id,
        "strategy": strategy,
        "agents": claimants,
        "responded": responded,
        "aggregated": aggregated,
        "handled": bool(aggregated),
    }
```

```python
# ── Backward Compatibility ────────────────────────────────────────


def is_multi_domain(query: str) -> bool:
    """Check if query requires multiple agents. Delegates to _get_claimants()."""
    return len(_get_claimants(query)) > 1


def identify_agents(query: str) -> list:
    """Identify which agent(s) should handle this query. Delegates to _get_claimants()."""
    claimants = _get_claimants(query)
    # Fallback: if no claimants but query matches common multi-domain triggers,
    # activate the three primary agents (preserves existing test behavior).
    if not claimants:
        triggers = ["como esta todo", "como va todo", "estado general",
                    "resumen completo", "que deberia saber", "que paso hoy",
                    "update", "dame el status"]
        if any(t in query.lower() for t in triggers):
            return ["infra", "personal", "finance"]
    return claimants


def list_agents() -> list:
    """List all agents with their status."""
    agents = []
    for name in AGENT_MODULES:
        mod = _load_agent(name)
        status = "healthy" if mod is not None else "unavailable"
        preferred_role = getattr(mod, "PREFERRED_ROLE", "fast") if mod else "unknown"
        agents.append({
            "name": name,
            "module": AGENT_MODULES[name],
            "status": status,
            "preferred_role": preferred_role,
            "has_owns": mod is not None and hasattr(mod, "owns"),
            "has_handle": mod is not None and hasattr(mod, "handle"),
        })
    return agents


def resolve_conflict(proposals: list) -> dict:
    """Delegate conflict resolution to arbitrator."""
    try:
        from agent_arbitrator import resolve
        return resolve(proposals)
    except Exception as e:
        return {"winner": None, "error": str(e)[:200]}
```
